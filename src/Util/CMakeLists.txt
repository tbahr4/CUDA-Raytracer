# Gather files at this level
file (GLOB LOCAL_SOURCE_FILES
	CONFIGURE_DEPENDS
	${CMAKE_CURRENT_SOURCE_DIR}/*.cpp
)

target_sources(${TARGET_NAME} PRIVATE ${LOCAL_SOURCE_FILES})

# Propagate source files to parent
list(APPEND SOURCE_FILES ${LOCAL_SOURCE_FILES})
set(SOURCE_FILES ${SOURCE_FILES} PARENT_SCOPE)
